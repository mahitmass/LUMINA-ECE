// File: frontend/src/App.jsx
import React, { useState, useEffect, useRef, useCallback } from 'react';
import {
  LineChart,
  Line,
  XAxis,
  YAxis,
  CartesianGrid,
  ResponsiveContainer,
  ReferenceLine
} from 'recharts';
import { 
  ShieldAlert, 
  Radio, 
  Activity, 
  Sliders, 
  Wifi, 
  WifiOff, 
  CheckCircle2, 
  Volume2 
} from 'lucide-react';

const MAX_WINDOW_SIZE = 100; // 100 samples rolling oscilloscope window
const UI_REFRESH_RATE_MS = 50; // 20 FPS decoupled render loop for zero browser lag

export default function App() {
  // --- UI RENDER STATE ---
  const [chartData, setChartData] = useState(() =>
    Array.from({ length: MAX_WINDOW_SIZE }, (_, i) => ({
      index: i,
      variance_db: 0,
      raw_amplitude: 0,
    }))
  );
  const [systemState, setSystemState] = useState('empty'); // 'empty' | 'breathing' | 'fall'
  const [latchedFallAlert, setLatchedFallAlert] = useState(false); // Persists until manually dismissed per UI spec
  const [wsConnected, setWsConnected] = useState(false);
  const [latestMetrics, setLatestMetrics] = useState({ variance_db: 0.0, raw_amplitude: 0.0, packetRate: 0 });

  // --- DSP & THRESHOLD CONFIGURATION STATE ---
  const [wsUrl, setWsUrl] = useState('ws://localhost:8000/ws/radar');
  const [fallThreshold, setFallThreshold] = useState(30.0);
  const [occupancyThreshold, setOccupancyThreshold] = useState(2.0);
  const [plotRaw, setPlotRaw] = useState(false);

  // --- HIGH-FREQUENCY MUTABLE REFS (BYPASSES REACT RENDER ON 100HZ WS INGEST) ---
  const dataBufferRef = useRef(
    Array.from({ length: MAX_WINDOW_SIZE }, (_, i) => ({
      index: i,
      variance_db: 0,
      raw_amplitude: 0,
    }))
  );
  const latestPacketRef = useRef({ state: 'empty', variance_db: 0.0, raw_amplitude: 0.0 });
  const packetCounterRef = useRef(0);
  const lastRateCalcRef = useRef(performance.now());
  const wsRef = useRef(null);
  const fallThresholdRef = useRef(fallThreshold);
  const occupancyThresholdRef = useRef(occupancyThreshold);

  // Sync threshold refs so WebSocket callback always reads fresh values without re-binding
  useEffect(() => {
    fallThresholdRef.current = fallThreshold;
    occupancyThresholdRef.current = occupancyThreshold;
  }, [fallThreshold, occupancyThreshold]);

  // --- WEBSOCKET CONNECTION & AUTO-RECONNECT ENGINE ---
  const connectWebSocket = useCallback(() => {
    if (wsRef.current && wsRef.current.readyState === WebSocket.OPEN) {
      wsRef.current.close();
    }

    const ws = new WebSocket(wsUrl);
    wsRef.current = ws;

    ws.onopen = () => {
      setWsConnected(true);
    };

    ws.onmessage = (event) => {
      try {
        const payload = JSON.parse(event.data);
        const variance = Number(payload.variance_db) || 0;
        const rawAmp = Number(payload.raw_amplitude) || 0;

        // Evaluate client-side threshold overrides or respect backend state
        let evaluatedState = payload.state || 'empty';
        if (variance >= fallThresholdRef.current) {
          evaluatedState = 'fall';
        } else if (variance >= occupancyThresholdRef.current) {
          evaluatedState = 'breathing';
        } else {
          evaluatedState = 'empty';
        }

        // Immediately latch critical fall alert (per UI/UX spec: persists until dismissed)
        if (evaluatedState === 'fall') {
          setLatchedFallAlert(true);
        }

        // Push to mutable ring buffer (O(1) amortized array slice + push)
        const nextBuffer = dataBufferRef.current.slice(1);
        nextBuffer.push({
          index: (dataBufferRef.current[dataBufferRef.current.length - 1]?.index + 1) || 0,
          variance_db: variance,
          raw_amplitude: rawAmp,
        });
        dataBufferRef.current = nextBuffer;

        latestPacketRef.current = {
          state: evaluatedState,
          variance_db: variance,
          raw_amplitude: rawAmp,
        };
        packetCounterRef.current += 1;
      } catch (err) {
        console.error('Malformed CSI telemetry packet:', err);
      }
    };

    ws.onclose = () => {
      setWsConnected(false);
    };

    ws.onerror = () => {
      ws.close();
    };
  }, [wsUrl]);

  useEffect(() => {
    connectWebSocket();
    const reconnectTimer = setInterval(() => {
      if (!wsRef.current || wsRef.current.readyState === WebSocket.CLOSED) {
        connectWebSocket();
      }
    }, 3000);

    return () => {
      clearInterval(reconnectTimer);
      if (wsRef.current) wsRef.current.close();
    };
  }, [connectWebSocket]);

  // --- DECOUPLED 20HZ UI RENDER LOOP ---
  useEffect(() => {
    const renderInterval = setInterval(() => {
      const now = performance.now();
      const elapsedSec = (now - lastRateCalcRef.current) / 1000;
      let currentHz = latestMetrics.packetRate;

      if (elapsedSec >= 1.0) {
        currentHz = Math.round(packetCounterRef.current / elapsedSec);
        packetCounterRef.current = 0;
        lastRateCalcRef.current = now;
      }

      // Batch update React state at controlled 20 FPS
      setChartData([...dataBufferRef.current]);
      setSystemState(latestPacketRef.current.state);
      setLatestMetrics({
        variance_db: latestPacketRef.current.variance_db,
        raw_amplitude: latestPacketRef.current.raw_amplitude,
        packetRate: currentHz,
      });
    }, UI_REFRESH_RATE_MS);

    return () => clearInterval(renderInterval);
  }, [latestMetrics.packetRate]);

  // --- MANUAL DISMISS PROTOCOL ---
  const handleDismissAlert = () => {
    setLatchedFallAlert(false);
  };

  // Determine dynamic color scheme per UI/UX Spec
  const isEmergency = latchedFallAlert || systemState === 'fall';
  const strokeColor = isEmergency ? '#ef4444' : '#22d3ee'; // Red 500 vs Cyan 400

  return (
    <div className="min-h-screen bg-[#020617] text-slate-100 font-sans flex flex-col selection:bg-cyan-500 selection:text-slate-950">
      
      {/* CRITICAL FALL TRANSIENT BANNER (Injected at top of DOM tree per UI/UX Spec) */}
      {latchedFallAlert && (
        <div className="w-full bg-[#ef4444] text-white px-6 py-4 flex items-center justify-between shadow-[0_0_30px_rgba(239,68,68,0.6)] animate-pulse z-50 border-b-2 border-red-300">
          <div className="flex items-center space-x-4">
            <ShieldAlert className="w-8 h-8 stroke-[2.5]" />
            <div>
              <h2 className="text-xl font-black tracking-wider uppercase">
                CRITICAL FALL DETECTED
              </h2>
              <p className="text-xs font-mono text-red-100">
                TRANSIENT DIELECTRIC DISTURBANCE EXCEEDED {fallThreshold.toFixed(1)} dB THRESHOLD • ACTUATORS ENGAGED
              </p>
            </div>
          </div>
          <button
            onClick={handleDismissAlert}
            className="px-5 py-2 bg-slate-950 hover:bg-slate-900 text-white font-bold text-xs uppercase tracking-widest rounded border border-red-300 transition cursor-pointer"
          >
            Acknowledge & Reset Protocol
          </button>
        </div>
      )}

      {/* TOP NAVIGATION / TELEMETRY BAR */}
      <header className="border-b border-slate-800/80 px-8 py-5 flex items-center justify-between bg-slate-950/50">
        <div className="flex items-center space-x-3">
          <Radio className={`w-7 h-7 ${isEmergency ? 'text-[#ef4444] animate-ping' : 'text-[#22d3ee]'}`} />
          <div>
            <h1 className="text-2xl font-extrabold tracking-tight text-white">
              CSI Radar Station
            </h1>
            <p className="text-xs font-mono text-slate-400">
              PROJECT SENTINEL // 802.11n HT20 MULTIPATH DOPPLER ENGINE
            </p>
          </div>
        </div>

        {/* Hardware & Link Telemetry */}
        <div className="flex items-center space-x-6 font-mono text-xs">
          <div className="flex items-center space-x-2 bg-slate-900/90 border border-slate-800 px-3 py-1.5 rounded">
            <Activity className="w-4 h-4 text-[#22d3ee]" />
            <span className="text-slate-400">SAMPLING:</span>
            <span className={`font-bold ${latestMetrics.packetRate >= 50 ? 'text-emerald-400' : 'text-amber-400'}`}>
              {latestMetrics.packetRate} Hz
            </span>
          </div>

          <div className="flex items-center space-x-2 bg-slate-900/90 border border-slate-800 px-3 py-1.5 rounded">
            {wsConnected ? (
              <>
                <Wifi className="w-4 h-4 text-emerald-400" />
                <span className="text-emerald-400 font-bold">WS LINK ONLINE</span>
              </>
            ) : (
              <>
                <WifiOff className="w-4 h-4 text-[#ef4444]" />
                <span className="text-[#ef4444] font-bold">WS DISCONNECTED</span>
              </>
            )}
          </div>
        </div>
      </header>

      {/* MAIN 4-COLUMN DESKTOP GRID ARCHITECTURE */}
      <main className="flex-1 grid grid-cols-1 lg:grid-cols-4 gap-6 p-8 max-w-[1800px] w-full mx-auto">
        
        {/* LEFT RAIL (COLUMN 1): STATUS READOUT & DSP FILTER CONFIG */}
        <div className="lg:col-span-1 flex flex-col justify-between space-y-6">
          
          {/* Top Panel: Room Status & Live Numerical Readouts */}
          <div className={`p-6 rounded-lg border transition-colors duration-200 ${
            isEmergency 
              ? 'bg-red-950/20 border-[#ef4444]' 
              : 'bg-slate-900/40 border-slate-800'
          }`}>
            <span className="text-xs font-bold tracking-widest uppercase text-slate-400 block mb-3">
              VOLUMETRIC STATE READOUT
            </span>

            {/* State Pill */}
            <div className="mb-6">
              {isEmergency ? (
                <div className="inline-flex items-center space-x-2 px-4 py-2 rounded bg-[#ef4444]/20 border border-[#ef4444] text-[#ef4444]">
                  <ShieldAlert className="w-5 h-5" />
                  <span className="font-black tracking-wider uppercase text-sm">
                    FALL TRANSIENT
                  </span>
                </div>
              ) : systemState === 'breathing' ? (
                <div className="inline-flex items-center space-x-2 px-4 py-2 rounded bg-cyan-950/60 border border-[#22d3ee] text-[#22d3ee]">
                  <Activity className="w-5 h-5" />
                  <span className="font-bold tracking-wider uppercase text-sm">
                    OCCUPIED (MICRO-DOPPLER)
                  </span>
                </div>
              ) : (
                <div className="inline-flex items-center space-x-2 px-4 py-2 rounded bg-slate-800/80 border border-slate-700 text-slate-300">
                  <CheckCircle2 className="w-5 h-5 text-slate-400" />
                  <span className="font-bold tracking-wider uppercase text-sm">
                    NOMINAL / EMPTY
                  </span>
                </div>
              )}
            </div>

            {/* Primary Data Readouts (Monospace per Typography Spec) */}
            <div className="space-y-4 border-t border-slate-800/80 pt-4 font-mono">
              <div>
                <div className="text-xs text-slate-400 uppercase">Subcarrier Variance</div>
                <div className={`text-2xl font-bold ${isEmergency ? 'text-[#ef4444]' : 'text-[#22d3ee]'}`}>
                  Variance: {latestMetrics.variance_db.toFixed(1)} dB
                </div>
              </div>

              <div>
                <div className="text-xs text-slate-400 uppercase">Mean Subcarrier Magnitude</div>
                <div className="text-lg text-slate-200">
                  Amplitude: {latestMetrics.raw_amplitude.toFixed(1)} dB
                </div>
              </div>

              <div className="pt-2 flex items-center justify-between text-xs text-slate-400 border-t border-slate-800/60">
                <span className="flex items-center gap-1.5">
                  <Volume2 className="w-4 h-4" /> GPIO Actuators (FR-05):
                </span>
                <span className={`font-bold ${isEmergency ? 'text-[#ef4444]' : 'text-slate-500'}`}>
                  {isEmergency ? 'ACTIVE (5V HIGH)' : 'STANDBY (LOW)'}
                </span>
              </div>
            </div>
          </div>

          {/* Bottom Panel: DSP Filter & Threshold Configuration */}
          <div className="p-6 rounded-lg bg-slate-900/40 border border-slate-800 space-y-5">
            <div className="flex items-center space-x-2">
              <Sliders className="w-4 h-4 text-[#22d3ee]" />
              <span className="text-xs font-bold tracking-widest uppercase text-slate-400">
                DSP FILTER CONFIG
              </span>
            </div>

            {/* Fall Threshold Slider */}
            <div className="space-y-1.5">
              <div className="flex justify-between text-xs font-mono">
                <label className="text-slate-300">Fall Alert Threshold</label>
                <span className="text-[#ef4444] font-bold">{fallThreshold.toFixed(1)} dB</span>
              </div>
              <input
                type="range"
                min="5.0"
                max="80.0"
                step="0.5"
                value={fallThreshold}
                onChange={(e) => setFallThreshold(parseFloat(e.target.value))}
                className="w-full accent-[#ef4444] bg-slate-800 h-1.5 rounded cursor-pointer"
              />
            </div>

            {/* Occupancy / Breathing Threshold Slider */}
            <div className="space-y-1.5">
              <div className="flex justify-between text-xs font-mono">
                <label className="text-slate-300">Micro-Doppler Threshold</label>
                <span className="text-[#22d3ee] font-bold">{occupancyThreshold.toFixed(1)} dB</span>
              </div>
              <input
                type="range"
                min="0.5"
                max="15.0"
                step="0.5"
                value={occupancyThreshold}
                onChange={(e) => setOccupancyThreshold(parseFloat(e.target.value))}
                className="w-full accent-[#22d3ee] bg-slate-800 h-1.5 rounded cursor-pointer"
              />
            </div>

            {/* Signal Overlay Toggle */}
            <div className="pt-2 border-t border-slate-800 flex items-center justify-between">
              <span className="text-xs text-slate-300 font-mono">Overlay Raw Amplitude</span>
              <button
                onClick={() => setPlotRaw(!plotRaw)}
                className={`px-3 py-1 rounded text-xs font-mono font-bold transition ${
                  plotRaw 
                    ? 'bg-[#22d3ee] text-slate-950' 
                    : 'bg-slate-800 text-slate-400 hover:bg-slate-700'
                }`}
              >
                {plotRaw ? 'ENABLED' : 'DISABLED'}
              </button>
            </div>

            {/* WebSocket Endpoint Input */}
            <div className="pt-2 border-t border-slate-800 space-y-1.5">
              <label className="text-[11px] font-mono text-slate-400 uppercase block">
                FastAPI WebSocket URI
              </label>
              <div className="flex space-x-2">
                <input
                  type="text"
                  value={wsUrl}
                  onChange={(e) => setWsUrl(e.target.value)}
                  className="w-full bg-slate-950 border border-slate-800 rounded px-2.5 py-1 text-xs font-mono text-slate-200 focus:outline-none focus:border-[#22d3ee]"
                />
              </div>
            </div>

            {/* Filter Specs Readout */}
            <div className="bg-slate-950/80 p-3 rounded border border-slate-800/80 text-[11px] font-mono text-slate-400 space-y-1">
              <div>FILTER: 4th-Order Butterworth</div>
              <div>PASSBAND: 0.1 Hz – 5.0 Hz</div>
              <div>WINDOW: 100 Samples (20-pt Var)</div>
            </div>
          </div>

        </div>

        {/* RIGHT CANVAS (COLUMNS 2-4): REAL-TIME OSCILLOSCOPE */}
        <div className={`lg:col-span-3 rounded-lg border p-6 flex flex-col justify-between transition-colors duration-200 ${
          isEmergency 
            ? 'bg-red-950/10 border-[#ef4444]/80 shadow-[0_0_40px_rgba(239,68,68,0.15)]' 
            : 'bg-slate-900/30 border-slate-800'
        }`}>
          
          {/* Canvas Header */}
          <div className="flex items-center justify-between mb-4">
            <div>
              <span className="text-xs font-bold tracking-widest uppercase text-slate-400 block">
                LIVE SUBCARRIER VARIANCE
              </span>
              <p className="text-xs font-mono text-slate-500 mt-0.5">
                Real-Time Multipath Distortion Energy (1.0s Sliding Variance Window)
              </p>
            </div>

            <div className="flex items-center space-x-6 text-xs font-mono">
              <div className="flex items-center space-x-2">
                <span className="w-3 h-0.5 bg-[#22d3ee] inline-block"></span>
                <span className="text-slate-300">Filtered Variance (dB)</span>
              </div>
              {plotRaw && (
                <div className="flex items-center space-x-2">
                  <span className="w-3 h-0.5 bg-slate-500 inline-block"></span>
                  <span className="text-slate-400">Raw CSI Amplitude</span>
                </div>
              )}
              <div className="flex items-center space-x-2">
                <span className="w-3 h-0.5 border-b border-dashed border-[#ef4444] inline-block"></span>
                <span className="text-[#ef4444]">Fall Limit ({fallThreshold} dB)</span>
              </div>
            </div>
          </div>

          {/* Recharts High-Frequency Oscilloscope Container */}
          <div className="flex-1 w-full min-h-[480px]">
            <ResponsiveContainer width="100%" height="100%">
              <LineChart
                data={chartData}
                margin={{ top: 20, right: 25, left: 0, bottom: 10 }}
              >
                <CartesianGrid 
                  strokeDasharray="3 3" 
                  stroke="#1e293b" 
                  vertical={false} 
                />
                <XAxis 
                  dataKey="index" 
                  stroke="#475569" 
                  tick={{ fill: '#64748b', fontSize: 11, fontFamily: 'monospace' }}
                  tickLine={false}
                />
                <YAxis 
                  domain={[0, (dataMax) => Math.max(40, Math.ceil(dataMax * 1.15))]} 
                  stroke="#475569"
                  tick={{ fill: '#64748b', fontSize: 11, fontFamily: 'monospace' }}
                  tickLine={false}
                  width={45}
                />

                {/* Calibrated Fall Threshold Reference Line */}
                <ReferenceLine
                  y={fallThreshold}
                  stroke="#ef4444"
                  strokeDasharray="6 4"
                  strokeWidth={1.5}
                />

                {/* Micro-Doppler Breathing Reference Line */}
                <ReferenceLine
                  y={occupancyThreshold}
                  stroke="#0891b2"
                  strokeDasharray="3 3"
                  strokeWidth={1}
                />

                {/* Optional Raw Amplitude Overlay */}
                {plotRaw && (
                  <Line
                    type="monotone"
                    dataKey="raw_amplitude"
                    stroke="#64748b"
                    strokeWidth={1.5}
                    dot={false}
                    isAnimationActive={false}
                  />
                )}

                {/* Primary Filtered CSI Subcarrier Variance Waveform */}
                <Line
                  type="monotone"
                  dataKey="variance_db"
                  stroke={strokeColor}
                  strokeWidth={2.5}
                  dot={false}
                  isAnimationActive={false} // CRITICAL: Must be false for 50-100Hz streaming
                />
              </LineChart>
            </ResponsiveContainer>
          </div>

          {/* Oscilloscope Footer Bar */}
          <div className="mt-4 pt-3 border-t border-slate-800/80 flex items-center justify-between text-[11px] font-mono text-slate-500">
            <div>CHANNEL: 6 (2.437 GHz) • BANDWIDTH: 20 MHz (52 L-LTF / 64 HT-LTF SUBCARRIERS)</div>
            <div>RENDER ENGINE: 20 FPS DECOUPLED RING-BUFFER • LATENCY BUDGET: &lt;400ms</div>
          </div>
        </div>

      </main>
    </div>
  );
}
